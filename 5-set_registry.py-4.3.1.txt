#!/usr/bin/env python3

from pyModbusTCP.client import ModbusClient
import sys

def main():
    if len(sys.argv) != 4:
        print("Użycie: python3 set_registry.py <adres_IP> <numer_rejestru> <wartość>")
        sys.exit(1)

    host = sys.argv[1]
    try:
        reg_addr = int(sys.argv[2])
        reg_value = int(sys.argv[3])
    except ValueError:
        print("Numer rejestru i wartość muszą być liczbami całkowitymi.")
        sys.exit(1)

    client = ModbusClient(host=host, port=502, auto_open=True, auto_close=True)

    if client.write_single_register(reg_addr, reg_value):
        print(f"✅ Rejestr {reg_addr} ustawiony na wartość {reg_value} na hoście {host}")
    else:
        print(f"❌ Błąd podczas zapisu do rejestru {reg_addr} na hoście {host}")

if __name__ == "__main__":
    main()
